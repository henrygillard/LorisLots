    --- PART A: Create the script ---

    Create `scripts/` directory if it doesn't exist, then write `scripts/phase-1-verify.sh` with the EXACT contents below. Make it executable (`chmod +x`).

    ```bash
    #!/bin/bash
    # Phase 1 verification — all 8 checks must pass, plus shopify theme check zero errors.
    # Source: .planning/phases/01-scaffold-design-system/01-VALIDATION.md
    # Run from repo root: bash scripts/phase-1-verify.sh
    set -u
    ROOT="."
    FAIL=0

    pass() { echo "[ OK ] $*"; }
    fail() { echo "[FAIL] $*"; FAIL=$((FAIL + 1)); }

    echo "=== Phase 1 Verification ==="

    # 1. {% include %} tags — SCAFFOLD-03
    COUNT=$(grep -rn "{%[-]*\s*include " --include="*.liquid" "$ROOT" 2>/dev/null | wc -l | tr -d ' ')
    if [ "$COUNT" = "0" ]; then pass "1. {% include %} tags: 0"; else fail "1. {% include %} tags: $COUNT (must be 0)"; fi

    # 2. lazysizes refs — SCAFFOLD-03
    COUNT=$(grep -rn "lazysizes" --include="*.liquid" --include="*.js" "$ROOT" 2>/dev/null | wc -l | tr -d ' ')
    if [ "$COUNT" = "0" ]; then pass "2. lazysizes refs: 0"; else fail "2. lazysizes refs: $COUNT (must be 0)"; fi

    # 3. console.log calls — SCAFFOLD-03
    COUNT=$(grep -rn "console\.log" --include="*.js" "$ROOT" 2>/dev/null | wc -l | tr -d ' ')
    if [ "$COUNT" = "0" ]; then pass "3. console.log calls: 0"; else fail "3. console.log calls: $COUNT (must be 0)"; fi

    # 4. Hardcoded hex in .liquid (excluding settings_schema.json which legitimately has hex defaults) — SCAFFOLD-03
    COUNT=$(grep -rn "#[0-9a-fA-F]\{3,6\}" --include="*.liquid" "$ROOT" 2>/dev/null | grep -v "config/settings_schema.json" | wc -l | tr -d ' ')
    # Dawn itself ships some hex in its own .liquid sections; we report rather than fail at zero.
    # In Phase 1 we have not migrated Dawn sections to var(--ll-*); that happens per-section in Phases 2-5.
    echo "[INFO] 4. Hardcoded hex in .liquid (Dawn sections, excl. settings_schema): $COUNT (informational; migrated per-section in Phases 2-5)"

    # 5. Legacy .liquid templates — SCAFFOLD-03 (templates/customers/ is allowed)
    COUNT=$(find "$ROOT/templates" -name "*.liquid" -not -path "*/customers/*" 2>/dev/null | wc -l | tr -d ' ')
    if [ "$COUNT" = "0" ]; then pass "5. Legacy .liquid templates (excl. customers/): 0"; else fail "5. Legacy .liquid templates: $COUNT (must be 0)"; fi

    # 6. ll-component.js exists — SCAFFOLD-04
    if [ -f "$ROOT/assets/ll-component.js" ]; then pass "6. ll-component.js exists"; else fail "6. ll-component.js missing"; fi

    # 7. Token injection in theme.liquid — SCAFFOLD-02
    if grep -q "ll-color-bg" "$ROOT/layout/theme.liquid" 2>/dev/null; then pass "7. --ll-color-bg present in layout/theme.liquid"; else fail "7. --ll-color-bg missing in layout/theme.liquid"; fi

    # 8. customElements.define guard — SCAFFOLD-04
    if grep -q "customElements.get" "$ROOT/assets/ll-component.js" 2>/dev/null; then pass "8. customElements.get guard present"; else fail "8. customElements.get guard missing"; fi

    echo ""
    echo "=== Static checks: $((8 - FAIL))/8 passed (excluding informational #4) ==="
    echo ""
    echo "=== shopify theme check ==="
    if command -v shopify >/dev/null 2>&1; then
      shopify theme check
      THEME_CHECK_RC=$?
      if [ "$THEME_CHECK_RC" = "0" ]; then
        pass "shopify theme check exit 0"
      else
        fail "shopify theme check exit $THEME_CHECK_RC"
      fi
    else
      echo "[WARN] shopify CLI not in PATH; skipping theme check"
    fi

    echo ""
    if [ "$FAIL" = "0" ]; then
      echo "=== PHASE 1 VERIFY: GREEN ==="
      exit 0
    else
      echo "=== PHASE 1 VERIFY: $FAIL FAILURE(S) ==="
      exit 1
    fi
    ```
