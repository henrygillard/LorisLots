Create the `assets/ll-component.js` skeleton that establishes the Web Components pattern every interactive Loris Lots component will follow. This file is both a working component (registering `<ll-component>`) and a documented template that Phase 2-5 plans copy as their starting point for `<ll-cart-drawer>`, `<ll-product-card>`, `<ll-pill-filter>`, etc.

Without a documented skeleton, each phase will reinvent the pattern slightly differently (different selector conventions, different event-binding patterns, missing `disconnectedCallback` cleanup, missing `customElements.get` guards). The skeleton enforces consistency by being the literal copy-paste source.

- `customElements.define` guarded by `customElements.get`
- Documented comments showing how to dispatch (`document.dispatchEvent(new CustomEvent('ll:cart:open', ...))`) and listen (`document.addEventListener`)

    ```javascript
    // assets/ll-component.js
    //
    // Loris Lots Web Component skeleton. THIS IS THE TEMPLATE.
    // Every new interactive Loris Lots component (ll-cart-drawer, ll-product-card,
    // ll-pill-filter, etc.) starts as a copy of this file with two find/replace passes:
    //   1. ll-component  -> ll-{your-name}        (custom element tag)
    //   2. LlComponent   -> Ll{YourName}          (class name; Pascal case)
    //
    // Architecture decisions encoded here (do NOT deviate without changing CONTEXT.md):
    //   - D-06: Web Components API (customElements.define) for all interactive components.
    //   - D-07: ll: events dispatch on document and listen on document. No singleton bus.
    //   - D-08: This file is the canonical template. Phases 2-5 copy it.
    //   - Pitfall 4 (RESEARCH.md): Guard customElements.define with customElements.get
    //     so multi-instance section rendering does not throw NotSupportedError.
    //
    // Event-naming convention (RFC-ish):
    //   ll:{domain}:{verb}     e.g., ll:cart:open, ll:cart:updated, ll:filter:applied
    //
    // Attribute-selector convention:
    //   data-ll-{purpose}      e.g., data-ll-trigger, data-ll-target, data-ll-state
    //   No inline onclick=, onchange=, etc. (violates SCAFFOLD-04).

    class LlComponent extends HTMLElement {
      connectedCallback() {
        // Wire DOM interactivity via data-ll-* selectors. Never inline onclick.
        this.trigger = this.querySelector('[data-ll-trigger]');
        if (this.trigger) {
          this._onTriggerClick = this.handleClick.bind(this);
          this.trigger.addEventListener('click', this._onTriggerClick);
        }

        // Listen for ll: events from anywhere in the document. No singleton bus —
        // document is the universal bus (D-07). Always bind once and store the
        // bound reference so disconnectedCallback can remove it.
        this._onCartOpen = this.handleCartOpen.bind(this);
        document.addEventListener('ll:cart:open', this._onCartOpen);

        // Example: another listener to watch cart updates. Uncomment when needed.
        // this._onCartUpdated = this.handleCartUpdated.bind(this);
        // document.addEventListener('ll:cart:updated', this._onCartUpdated);
      }

      disconnectedCallback() {
        // ALWAYS clean up document-level listeners. They outlive the element
        // because document persists for the life of the page.
        if (this.trigger && this._onTriggerClick) {
          this.trigger.removeEventListener('click', this._onTriggerClick);
        }
        document.removeEventListener('ll:cart:open', this._onCartOpen);
        // document.removeEventListener('ll:cart:updated', this._onCartUpdated);
      }

      handleClick(event) {
        // Dispatch an ll: event on document so any listener (regardless of DOM
        // location) can react. bubbles: true is harmless on document but useful
        // if a future refactor dispatches from a child element.
        document.dispatchEvent(new CustomEvent('ll:cart:open', {
          bubbles: true,
          detail: { source: 'll-component', triggeredBy: event.currentTarget }
        }));
      }

      handleCartOpen(event) {
        // React to ll:cart:open dispatched by any other component.
        // event.detail contains whatever the dispatcher passed.
        // Default: no-op. Subclasses / copies override.
      }

      // handleCartUpdated(event) {
      //   // React to ll:cart:updated. Uncomment listener registration above.
      // }
    }

    // Pitfall 4: guard against duplicate-registration NotSupportedError when a
    // section that includes this script is rendered more than once on a page.
    if (!customElements.get('ll-component')) {
      customElements.define('ll-component', LlComponent);
    }
    ```
